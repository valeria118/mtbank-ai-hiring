# Опциональный бэкенд диаризации по голосу — pyannote.audio.
# Ставится ПОВЕРХ requirements.txt и только при сборке с WITH_PYANNOTE=true
# (см. docker-compose.yml). Без него DIARIZATION_BACKEND=auto продолжает
# работать на эвристике по паузам: asr/diarizer.py ловит ImportError и молча
# откатывается. Режим DIARIZATION_BACKEND=pyannote в таком образе упадёт с
# ошибкой — это осознанно, явный выбор бэкенда не должен подменяться тихо.
#
# Колёса берём с CPU-индекса PyTorch, а не с PyPI: там torch тянет за собой
# весь CUDA-стек (nvidia-cublas/cudnn/cufft/cusolver/nccl/cusparselt + triton)
# — 2.7 ГБ колёс против ~300 МБ. GPU здесь не нужен: на CUDA в этом проекте
# считает faster-whisper, а он работает через ctranslate2, не через torch.
--extra-index-url https://download.pytorch.org/whl/cpu

# torchaudio больше не развивается отдельно от torch (последний релиз —
# 2.11.0, torch тем временем ушёл на более новые версии) — версии закреплены
# строго парой, иначе pip тянет несовместимые torch/torchaudio и pyannote
# падает на импорте (AttributeError: module 'torchaudio' has no attribute
# 'AudioMetaData').
torch==2.11.0
torchaudio==2.11.0
# torchvision мы напрямую не используем, но образ ghcr.io/open-webui/pipelines
# уже содержит его (собранный под torch==2.8.0, под примеры пайплайнов из
# коробки OpenWebUI) — при обновлении torch до 2.11.0 старый torchvision
# рассинхронизируется и падает с "partially initialized module 'torchvision'
# has no attribute 'extension' (circular import)", как только что-то в цепочке
# импортов pyannote его затрагивает (обнаружено при живом запуске). Явно
# закрепляем версию, требующую ровно torch==2.11.0, чтобы pip поставил
# совместимую, а не оставил старую от базового образа.
torchvision==0.26.0
pyannote.audio==4.0.7
